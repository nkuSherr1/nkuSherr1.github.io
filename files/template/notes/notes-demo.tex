\section{Environments}

\begin{definition}[边密度]
设 $X,Y$ 为图 $G$ 的顶点子集。边密度定义为
\[
d_G(X,Y)=\frac{e_G(X,Y)}{|X|\,|Y|}.
\]
\end{definition}

\begin{theorem}[正则性引理]
对每个 $\varepsilon>0$，存在 $M=M(\varepsilon)$，使得每个图都有一个至多 $M$ 块的 $\varepsilon$-正则剖分。
\end{theorem}

\begin{lemma}
能量在加细下单调不减。
\end{lemma}

\begin{proposition}
不规则对的能量提升至少为 $\varepsilon^5$。
\end{proposition}

\begin{corollary}
塔型上界不能改成 $\varepsilon$ 的多项式。
\end{corollary}

\begin{example}
完全二部图的每一对分部都是 $0$-正则的。
\end{example}

\begin{remark}
同一份笔记里，$\varepsilon$ 既控制子集大小，也控制密度偏差。
\end{remark}
