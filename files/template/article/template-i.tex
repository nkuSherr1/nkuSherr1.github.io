\section{Gallery}
\begin{thmm}{}
A finite graph admits an $\varepsilon$-regular partition with at most $M(\varepsilon)$ parts.
\end{thmm}
\begin{deff}{}
The edge density of $X,Y\subseteq V(G)$ is $d(X,Y)=e(X,Y)/(|X||Y|)$.
\end{deff}
\begin{eg}{}
Complete bipartite graphs are $0$-regular pairs.
\end{eg}
\begin{lemm}{}
Energy is monotone under refinement.
\end{lemm}
\begin{proo}{}
Energy is a convex combination of squared densities.
\end{proo}
\begin{coll}{}
The tower-type bound cannot be replaced by a polynomial in $\varepsilon^{-1}$.
\end{coll}
\begin{ex}{}
Compute $d(X,Y)$ for a random bipartite graph.
\end{ex}
\begin{pf}
Energy increases by at least $\varepsilon^5$ at each irregular step.
\end{pf}
\begin{sol}
Partition, refine, and stop when every pair is regular.
\end{sol}
\begin{rem}
Keep a single $\varepsilon$ for both size and density deviation.
\end{rem}
\begin{ana}
The increment comes from the variance of a random edge density.
\end{ana}
\begin{cla}
Empty blocks do not contribute to the energy.
\end{cla}
