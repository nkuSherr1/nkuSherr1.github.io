\section{环境}

\begin{definition}[边密度]
设 $X,Y$ 为图 $G$ 的顶点子集。边密度定义为
\[
d_G(X,Y)=\frac{e_G(X,Y)}{|X|\,|Y|}.
\]
\end{definition}

\begin{theorem}[正则性引理]
对每个 $\varepsilon>0$，存在 $M=M(\varepsilon)$，使得每个图都有一个至多 $M$ 块的 $\varepsilon$-正则剖分。
\end{theorem}
